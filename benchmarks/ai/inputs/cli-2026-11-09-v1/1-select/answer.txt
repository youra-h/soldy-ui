```vue
<script setup>
import { computed, nextTick, onBeforeUnmount, onMounted, ref, useId, watch } from 'vue'

const props = defineProps({
  items: {
    type: Array,
    default: () => [],
  },
  rows: {
    type: Array,
    default: () => [],
  },
  locale: {
    type: String,
    default: 'en',
    validator: value => ['en', 'ru', 'ar'].includes(value),
  },
  dir: {
    type: String,
    default: 'ltr',
    validator: value => ['ltr', 'rtl'].includes(value),
  },
})

const uid = useId()
const inputId = `${uid}-input`
const listId = `${uid}-list`
const hintId = `${uid}-hint`

const root = ref(null)
const input = ref(null)
const listbox = ref(null)
const open = ref(false)
const composing = ref(false)
const query = ref('')
const selectedValues = ref([])
const activeValue = ref(null)

const selectedSet = computed(() => new Set(selectedValues.value))
const itemLabels = computed(
  () => new Map(props.items.map(item => [item.value, item.text])),
)
const selectedItems = computed(() =>
  selectedValues.value.map(value => ({
    value,
    text: itemLabels.value.get(value) ?? value,
  })),
)

const filteredItems = computed(() => {
  const search = query.value.trim().toLowerCase()
  return search
    ? props.items.filter(item => item.text.toLowerCase().includes(search))
    : props.items
})

const activeIndex = computed(() =>
  filteredItems.value.findIndex(item => item.value === activeValue.value),
)

const activeId = computed(() =>
  open.value && activeIndex.value >= 0
    ? `${uid}-option-${activeIndex.value}`
    : undefined,
)

const selectionMessage = computed(() => {
  const count = selectedValues.value.length
  return `${count} ${count === 1 ? 'item' : 'items'} selected.`
})

function preferredItem() {
  const search = query.value.trim().toLowerCase()
  if (search) {
    return (
      filteredItems.value.find(item =>
        item.text.toLowerCase().startsWith(search),
      ) ?? filteredItems.value[0]
    )
  }
  return filteredItems.value[0]
}

function activatePreferred() {
  activeValue.value = preferredItem()?.value ?? null
}

async function revealActive() {
  await nextTick()
  if (!open.value || !listbox.value || activeIndex.value < 0) return

  const option = listbox.value.children[activeIndex.value]
  if (!option) return

  const containerBounds = listbox.value.getBoundingClientRect()
  const optionBounds = option.getBoundingClientRect()
  const top = containerBounds.top + listbox.value.clientTop
  const bottom = top + listbox.value.clientHeight

  if (optionBounds.top < top) {
    listbox.value.scrollTop -= top - optionBounds.top
  } else if (optionBounds.bottom > bottom) {
    listbox.value.scrollTop += optionBounds.bottom - bottom
  }
}

function openPopup() {
  if (!open.value) {
    open.value = true
    if (activeIndex.value < 0) activatePreferred()
    revealActive()
  }
}

function closePopup() {
  open.value = false
}

function focusInput() {
  input.value?.focus({ preventScroll: true })
}

function toggleValue(value) {
  if (composing.value) return

  selectedValues.value = selectedSet.value.has(value)
    ? selectedValues.value.filter(selected => selected !== value)
    : [...selectedValues.value, value]
}

function removeValue(value) {
  if (composing.value) return
  selectedValues.value = selectedValues.value.filter(selected => selected !== value)
  focusInput()
}

function chooseItem(item) {
  if (composing.value) return
  activeValue.value = item.value
  toggleValue(item.value)
  focusInput()
}

function updateQuery(value) {
  query.value = value
  open.value = true
  activatePreferred()
  revealActive()
}

function onInput(event) {
  if (composing.value || event.isComposing) return
  updateQuery(event.target.value)
}

function onCompositionEnd(event) {
  composing.value = false
  updateQuery(event.target.value)
}

function onKeydown(event) {
  if (composing.value || event.isComposing || event.keyCode === 229) return

  if (event.key === 'Tab') {
    closePopup()
    return
  }

  // Preserve native editing shortcuts, Home/End, and Space.
  if (event.ctrlKey || event.metaKey || event.altKey || event.shiftKey) return

  if (event.key === 'ArrowDown' || event.key === 'ArrowUp') {
    event.preventDefault()

    const wasOpen = open.value
    open.value = true
    const length = filteredItems.value.length

    if (!length) {
      activeValue.value = null
      return
    }

    let index = activeIndex.value
    if (!wasOpen || index < 0) {
      index = event.key === 'ArrowDown' ? 0 : length - 1
    } else {
      index = Math.max(
        0,
        Math.min(length - 1, index + (event.key === 'ArrowDown' ? 1 : -1)),
      )
    }

    activeValue.value = filteredItems.value[index].value
    revealActive()
  } else if (event.key === 'Enter') {
    event.preventDefault()

    if (!open.value) {
      openPopup()
    } else if (!event.repeat && activeIndex.value >= 0) {
      toggleValue(filteredItems.value[activeIndex.value].value)
    }
  }
}

function onEscape(event) {
  if (composing.value || event.isComposing || event.keyCode === 229) return
  if (!open.value) return

  event.preventDefault()
  event.stopPropagation()
  closePopup()
  focusInput()
}

function togglePopup() {
  if (composing.value) return
  if (open.value) closePopup()
  else openPopup()
  focusInput()
}

function onOutsidePointer(event) {
  if (root.value && !root.value.contains(event.target)) closePopup()
}

function onFocusout(event) {
  if (!root.value?.contains(event.relatedTarget)) closePopup()
}

watch(filteredItems, () => {
  if (activeIndex.value < 0) {
    activatePreferred()
    if (open.value) revealActive()
  }
})

onMounted(() => {
  document.addEventListener('pointerdown', onOutsidePointer)
})

onBeforeUnmount(() => {
  document.removeEventListener('pointerdown', onOutsidePointer)
})
</script>

<template>
  <div
    ref="root"
    class="select"
    :dir="props.dir"
    :lang="props.locale"
    @keydown.esc="onEscape"
    @focusout="onFocusout"
  >
    <label class="label" :for="inputId">Choose items</label>

    <ul v-if="selectedItems.length" class="selections" aria-label="Selected items">
      <li v-for="item in selectedItems" :key="item.value" class="chip">
        <span class="chip-text">{{ item.text }}</span>
        <button
          type="button"
          class="remove"
          :aria-label="`Remove ${item.text}`"
          :disabled="composing"
          @click="removeValue(item.value)"
        >
          <span aria-hidden="true">×</span>
        </button>
      </li>
    </ul>

    <input
      v-for="value in selectedValues"
      :key="value"
      type="hidden"
      name="items"
      :value="value"
    />

    <div class="field">
      <input
        :id="inputId"
        ref="input"
        class="search"
        type="text"
        role="combobox"
        aria-autocomplete="list"
        aria-haspopup="listbox"
        :aria-expanded="open"
        :aria-controls="open ? listId : undefined"
        :aria-activedescendant="activeId"
        :aria-describedby="hintId"
        :value="query"
        autocomplete="off"
        autocapitalize="off"
        :spellcheck="false"
        placeholder="Search items…"
        @click="openPopup"
        @input="onInput"
        @keydown="onKeydown"
        @compositionstart="composing = true"
        @compositionend="onCompositionEnd"
      />

      <button
        type="button"
        class="toggle"
        :aria-label="open ? 'Close options' : 'Open options'"
        :aria-expanded="open"
        :aria-controls="open ? listId : undefined"
        :disabled="composing"
        @mousedown.prevent
        @click="togglePopup"
      >
        <svg
          viewBox="0 0 20 20"
          width="20"
          height="20"
          fill="none"
          aria-hidden="true"
          :class="{ rotated: open }"
        >
          <path
            d="m5 7.5 5 5 5-5"
            stroke="currentColor"
            stroke-width="1.8"
            stroke-linecap="round"
            stroke-linejoin="round"
          />
        </svg>
      </button>
    </div>

    <div v-if="open" class="popup">
      <!-- Every matching option remains reachable through native scrolling. -->
      <ul
        :id="listId"
        ref="listbox"
        class="options"
        role="listbox"
        aria-label="Choose items"
        aria-multiselectable="true"
      >
        <li
          v-for="(item, index) in filteredItems"
          :id="`${uid}-option-${index}`"
          :key="item.value"
          role="option"
          class="option"
          :class="{
            active: activeValue === item.value,
            selected: selectedSet.has(item.value),
          }"
          :aria-selected="selectedSet.has(item.value)"
          @mousedown.prevent
          @click="chooseItem(item)"
        >
          <span class="check" aria-hidden="true">
            {{ selectedSet.has(item.value) ? '✓' : '' }}
          </span>
          <span class="option-text">{{ item.text }}</span>
        </li>
      </ul>

      <p v-if="!filteredItems.length" class="empty" role="status">
        No matching items.
      </p>
    </div>

    <p :id="hintId" class="hint">
      Type to filter. Use ↑ and ↓ to browse, Enter to select or deselect, and
      Escape to close.
    </p>
    <p class="selection-status" role="status" aria-live="polite" aria-atomic="true">
      {{ selectionMessage }}
    </p>
  </div>
</template>
```

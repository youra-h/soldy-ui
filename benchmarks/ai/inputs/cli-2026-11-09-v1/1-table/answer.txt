```vue
<script setup>
import { computed, ref, watch } from 'vue'

const props = defineProps({
  items: { type: Array, default: () => [] },
  rows: { type: Array, default: () => [] },
  locale: { type: String, default: 'en' },
  dir: { type: String, default: 'ltr' }
})

const translations = {
  en: {
    name: 'Name',
    score: 'Score',
    selection: 'Selection',
    selectAll: 'Select all records across all pages',
    selectRow: 'Select record',
    id: 'ID',
    ascending: 'Ascending',
    descending: 'Descending',
    unsorted: 'Not sorted',
    sortAscending: 'Sort ascending',
    sortDescending: 'Sort descending',
    selected: 'Selected',
    clear: 'Clear selection',
    first: 'First',
    previous: 'Previous',
    next: 'Next',
    last: 'Last',
    pagination: 'Records pagination',
    page: 'Page',
    of: 'of',
    showing: 'Showing',
    empty: 'No records',
    scroll: 'Records table, scrollable',
    help: '100 records per page. Scroll within the table to view the current page.'
  },
  ru: {
    name: 'Имя',
    score: 'Баллы',
    selection: 'Выбор',
    selectAll: 'Выбрать все записи на всех страницах',
    selectRow: 'Выбрать запись',
    id: 'ID',
    ascending: 'По возрастанию',
    descending: 'По убыванию',
    unsorted: 'Без сортировки',
    sortAscending: 'Сортировать по возрастанию',
    sortDescending: 'Сортировать по убыванию',
    selected: 'Выбрано',
    clear: 'Снять выделение',
    first: 'Первая',
    previous: 'Назад',
    next: 'Вперёд',
    last: 'Последняя',
    pagination: 'Страницы записей',
    page: 'Страница',
    of: 'из',
    showing: 'Показано',
    empty: 'Нет записей',
    scroll: 'Таблица Records с прокруткой',
    help: '100 записей на странице. Прокрутите таблицу для просмотра текущей страницы.'
  },
  ar: {
    name: 'الاسم',
    score: 'النتيجة',
    selection: 'التحديد',
    selectAll: 'تحديد جميع السجلات في جميع الصفحات',
    selectRow: 'تحديد السجل',
    id: 'المعرّف',
    ascending: 'تصاعدي',
    descending: 'تنازلي',
    unsorted: 'غير مرتب',
    sortAscending: 'ترتيب تصاعدي',
    sortDescending: 'ترتيب تنازلي',
    selected: 'المحدد',
    clear: 'مسح التحديد',
    first: 'الأولى',
    previous: 'السابقة',
    next: 'التالية',
    last: 'الأخيرة',
    pagination: 'صفحات السجلات',
    page: 'الصفحة',
    of: 'من',
    showing: 'عرض',
    empty: 'لا توجد سجلات',
    scroll: 'جدول Records قابل للتمرير',
    help: '100 سجل لكل صفحة. مرّر داخل الجدول لعرض الصفحة الحالية.'
  }
}

const language = computed(() =>
  Object.prototype.hasOwnProperty.call(translations, props.locale)
    ? props.locale
    : 'en'
)
const t = computed(() => translations[language.value])
const direction = computed(() => props.dir === 'rtl' ? 'rtl' : 'ltr')

const pageSize = 100
const requestedPage = ref(1)
const sortKey = ref('name')
const sortDirection = ref('ascending')
const selectedIds = ref(new Set())
const scrollArea = ref(null)

const availableIds = computed(() => new Set(props.rows.map(row => row.id)))

watch(availableIds, ids => {
  const remaining = new Set([...selectedIds.value].filter(id => ids.has(id)))
  if (remaining.size !== selectedIds.value.size) {
    selectedIds.value = remaining
  }
}, { flush: 'sync' })

// Copy before sorting. Code-point comparison avoids server/client ICU differences.
const sortedRows = computed(() => {
  const multiplier = sortDirection.value === 'ascending' ? 1 : -1

  return props.rows
    .map((row, index) => ({ row, index }))
    .sort((a, b) => {
      const left = a.row[sortKey.value]
      const right = b.row[sortKey.value]
      const comparison = left < right ? -1 : left > right ? 1 : 0
      return comparison * multiplier || a.index - b.index
    })
    .map(entry => entry.row)
})

const total = computed(() => props.rows.length)
const pageCount = computed(() => Math.max(1, Math.ceil(total.value / pageSize)))
const page = computed(() => Math.min(requestedPage.value, pageCount.value))
const start = computed(() => total.value ? (page.value - 1) * pageSize + 1 : 0)
const end = computed(() => Math.min(page.value * pageSize, total.value))
const visibleRows = computed(() =>
  sortedRows.value.slice((page.value - 1) * pageSize, page.value * pageSize)
)

const selectedCount = computed(() => selectedIds.value.size)
const allSelected = computed(() =>
  availableIds.value.size > 0 &&
  selectedCount.value === availableIds.value.size
)
const partlySelected = computed(() =>
  selectedCount.value > 0 && !allSelected.value
)

function resetScroll() {
  if (scrollArea.value) scrollArea.value.scrollTop = 0
}

function changePage(nextPage) {
  requestedPage.value = Math.max(1, Math.min(nextPage, pageCount.value))
  resetScroll()
}

function sortBy(key) {
  if (sortKey.value === key) {
    sortDirection.value =
      sortDirection.value === 'ascending' ? 'descending' : 'ascending'
  } else {
    sortKey.value = key
    sortDirection.value = 'ascending'
  }
  requestedPage.value = 1
  resetScroll()
}

function ariaSort(key) {
  return sortKey.value === key ? sortDirection.value : 'none'
}

function sortState(key) {
  return sortKey.value === key
    ? t.value[sortDirection.value]
    : t.value.unsorted
}

function sortButtonName(key) {
  const action =
    sortKey.value === key && sortDirection.value === 'ascending'
      ? t.value.sortDescending
      : t.value.sortAscending
  return `${t.value[key]}: ${action}`
}

function toggleRow(id, checked) {
  const next = new Set(selectedIds.value)
  if (checked) next.add(id)
  else next.delete(id)
  selectedIds.value = next
}

function toggleAll(checked) {
  selectedIds.value = checked ? new Set(availableIds.value) : new Set()
}

function clearSelection() {
  selectedIds.value = new Set()
}

function rowCheckboxName(row) {
  return `${t.value.selectRow}: ${row.name}; ${t.value.id}: ${row.id}`
}
</script>

<template>
  <section class="records" :lang="language" :dir="direction" aria-label="Records">
    <p class="help">{{ t.help }}</p>

    <div class="selection-toolbar">
      <p class="selection-summary">
        {{ t.selected }}: {{ selectedCount }} {{ t.of }} {{ total }}

        class="action"
        type="button"
        :disabled="selectedCount === 0"
        @click="clearSelection"
      >
        {{ t.clear }}
      </button>

    <div
      ref="scrollArea"

      role="region"
      :aria-label="t.scroll"
      tabindex="0"
    >
      <table>
        <caption>Records</caption>
        <colgroup>
          <col class="selection-column" />
          <col />
          <col class="score-column" />
        </colgroup>
        <thead>
          <tr>
            <th scope="col" class="checkbox-cell">
              <span class="sr-only">{{ t.selection }}</span>
              <label class="checkbox-target">
                <input
                  type="checkbox"
                  :checked="allSelected"
                  :indeterminate="partlySelected"
                  :disabled="total === 0"
                  @change="toggleAll($event.target.checked)"
                />
                <span class="sr-only">{{ t.selectAll }}</span>
              </label>
            </th>
            <th scope="col" :aria-sort="ariaSort('name')">
              <button
                type="button"
                class="sort-button"
                :aria-label="sortButtonName('name')"
                @click="sortBy('name')"
              >
                <span>{{ t.name }}</span>
                <span class="sort-state" aria-hidden="true">
                  {{ sortState('name') }}
                  <span v-if="sortKey === 'name'">
                    {{ sortDirection === 'ascending' ? '↑' : '↓' }}
                  </span>
                </span>
              </button>
            </th>
            <th scope="col" :aria-sort="ariaSort('score')">
              <button
                type="button"
                class="sort-button"
                :aria-label="sortButtonName('score')"
                @click="sortBy('score')"
              >
                <span>{{ t.score }}</span>
                <span class="sort-state" aria-hidden="true">
                  {{ sortState('score') }}
                  <span v-if="sortKey === 'score'">
                    {{ sortDirection === 'ascending' ? '↑' : '↓' }}
                  </span>
                </span>
              </button>
            </th>
          </tr>
        </thead>
        <tbody>
          <tr v-if="visibleRows.length === 0">
            <td colspan="3" class="empty">{{ t.empty }}</td>
          </tr>
          <tr
            v-for="row in visibleRows"
            :key="row.id"
            :class="{ selected: selectedIds.has(row.id) }"
          >
            <td class="checkbox-cell">
              <label class="checkbox-target">
                <input
                  type="checkbox"
                  :checked="selectedIds.has(row.id)"
                  @change="toggleRow(row.id, $event.target.checked)"
                />
                <span class="sr-only">{{ rowCheckboxName(row) }}</span>
              </label>
            </td>
            <th scope="row" class="record-name">
              <bdi>{{ row.name }}</bdi>
            </th>
            <td class="score"><bdi>{{ row.score }}</bdi></td>
          </tr>
        </tbody>
      </table>
    </div>

    <p class="page-summary">
      {{ t.showing }} {{ start }}–{{ end }} {{ t.of }} {{ total }}.
      {{ t.page }} {{ page }} {{ t.of }} {{ pageCount }}.
    </p>

    <nav class="pagination" :aria-label="t.pagination">
      <button
        type="button"
        class="action"
        :disabled="page === 1"
        @click="changePage(1)"
      >
        {{ t.first }}
      </button>
      <button
        type="button"
        class="action"
        :disabled="page === 1"
        @click="changePage(page - 1)"
      >
        {{ t.previous }}
      </button>
      <button
        type="button"
        class="action"
        :disabled="page === pageCount"
        @click="changePage(page + 1)"
      >
        {{ t.next }}
      </button>
      <button
        type="button"
        class="action"
        :disabled="page === pageCount"
        @click="changePage(pageCount)"
      >
        {{ t.last }}
      </button>
    </nav>

    <p class="sr-only" role="status" aria-atomic="true">
      {{ t.showing }} {{ start }}–{{ end }} {{ t.of }} {{ total }}.
      {{ t.page }} {{ page }} {{ t.of }} {{ pageCount }}.
      {{ t.selected }}: {{ selectedCount }}.
      {{ t[sortKey] }}: {{ t[sortDirection] }}.
    </p>
  </section>
</template>
```

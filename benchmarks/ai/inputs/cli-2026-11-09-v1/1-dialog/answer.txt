<script setup>
import { ref } from 'vue'

const props = defineProps({
  items: {
    type: Array,
    default: () => [],
  },
  rows: {
    type: Array,
    default: () => [],
  },
  locale: {
    type: String,
    default: 'en',
    validator: (value) => ['en', 'ru', 'ar'].includes(value),
  },
  dir: {
    type: String,
    default: 'ltr',
    validator: (value) => ['ltr', 'rtl'].includes(value),
  },
})

const dialog = ref(null)
const opener = ref(null)
const nameInput = ref(null)
const cancelButton = ref(null)
const saveButton = ref(null)

const savedName = ref('')
const draftName = ref('')

function openDialog() {
  if (!dialog.value || dialog.value.open) return

  draftName.value = savedName.value
  dialog.value.showModal()
  nameInput.value?.focus()
}

function closeDialog() {
  if (!dialog.value?.open) return

  dialog.value.close()
  opener.value?.focus()
}

function cancel() {
  draftName.value = savedName.value
  closeDialog()
}

function save() {
  savedName.value = draftName.value
  closeDialog()
}

function trapTab(event) {
  if (event.key !== 'Tab') return

  const controls = [
    nameInput.value,
    cancelButton.value,
    saveButton.value,
  ].filter(Boolean)

  if (!controls.length) return

  const activeElement = event.currentTarget.ownerDocument.activeElement
  const currentIndex = controls.indexOf(activeElement)
  const nextIndex =
    currentIndex === -1
      ? event.shiftKey
        ? controls.length - 1
        : 0
      : (currentIndex + (event.shiftKey ? -1 : 1) + controls.length) %
        controls.length

  event.preventDefault()
  controls[nextIndex].focus()
}
</script>

<template>
  <div class="settings" :dir="props.dir" lang="en">
    <button ref="opener" type="button" @click="openDialog">
      Open dialog
    </button>

    <dialog
      ref="dialog"
      class="dialog"
      aria-label="Settings"
      aria-modal="true"
      @cancel.prevent="cancel"
      @keydown="trapTab"
    >
      <form class="form" @submit.prevent="save">
        <h2>Settings</h2>

        <label class="field">
          <span>Name</span>
          <input
            ref="nameInput"
            v-model="draftName"
            :lang="props.locale"
            type="text"
            name="name"
            autocomplete="name"
          />
        </label>

        <div class="actions">
          <button ref="cancelButton" type="button" @click="cancel">
            Cancel
          </button>
          <button ref="saveButton" class="primary" type="submit">
            Save
          </button>
        </div>
      </form>
    </dialog>
  </div>
</template>
